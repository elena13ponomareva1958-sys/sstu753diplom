+class OfflineAnalysisViewModel(private val container: AppContainer) : ViewModel() {
+    private val _state = MutableStateFlow(OfflineAnalysisUiState())
+    val state = _state.asStateFlow()
+    private var analysisJob: Job? = null
+
+    fun analyze(uri: Uri, settings: AppSettings) {
+        analysisJob?.cancel()
+        analysisJob = viewModelScope.launch {
+            val name = queryDisplayName(uri)
+            val mime = container.appContext.contentResolver.getType(uri).orEmpty()
+            val isVideo = mime.startsWith("video/") || name.lowercase(Locale.ROOT).let {
+                it.endsWith(".mp4") || it.endsWith(".mkv") || it.endsWith(".avi") ||
+                    it.endsWith(".mov") || it.endsWith(".webm") || it.endsWith(".3gp")
+            }
+            _state.value = OfflineAnalysisUiState(
+                busy = true,
+                sourceName = name,
+                sourceType = when {
+                    isVideo -> "Видео"
+                    mime.startsWith("image/") -> "Изображение"
+                    else -> "Файл"
+                },
+                message = "Подготовка файла к проверке…",
+            )
+
+            try {
+                if (isVideo) {
+                    analyzeVideo(uri, name, settings)
+                } else {
+                    analyzeImage(uri, name, settings)
+                }
+            } catch (_: CancellationException) {
+                _state.value = _state.value.copy(
+                    busy = false,
+                    message = "Проверка остановлена",
+                    isError = false,
+                )
+            } catch (error: Throwable) {
+                _state.value = _state.value.copy(
+                    busy = false,
+                    isError = true,
+                    message = "Не удалось проверить файл: ${error.message ?: error.javaClass.simpleName}",
+                )
+            }
+        }
+    }
+
+    fun cancel() {
+        analysisJob?.cancel()
+    }
+
+    fun clear() {
+        analysisJob?.cancel()
+        replacePreview(null)
+        _state.value = OfflineAnalysisUiState()
+    }
+
+    private suspend fun analyzeImage(uri: Uri, name: String, settings: AppSettings) {
+        val result = withContext(Dispatchers.Default) {
+            val bitmap = decodeImage(uri)
+            try {
+                LiteRtYoloDetector(container.appContext).use { detector ->
+                    val inference = detector.detect(
+                        source = bitmap,
+                        settings = settings,
+                        confidenceThreshold = settings.confidenceThreshold,
+                        iouThreshold = settings.iouThreshold,
+                    )
+                    val preview = bitmap.copy(Bitmap.Config.ARGB_8888, true)
+                        ?: error("Не удалось подготовить изображение")
+                    DetectionFileStore.drawDetections(preview, inference.detections)
+                    if (settings.autoSaveDetections && inference.detections.isNotEmpty()) {
+                        saveToHistory(bitmap, inference.detections)
+                    }
+                    ImageResult(preview, inference.detections, inference.inferenceMs)
+                }
+            } finally {
+                bitmap.recycle()
+            }
+        }
+
+        replacePreview(result.preview)
+        _state.value = OfflineAnalysisUiState(
+            sourceName = name,
+            sourceType = "Изображение",
+            preview = result.preview,
+            processedFrames = 1,
+            totalFrames = 1,
+            defectFrames = if (result.detections.isEmpty()) 0 else 1,
+            detections = result.detections,
+            maxConfidence = result.detections.maxOfOrNull { it.confidence } ?: 0f,
